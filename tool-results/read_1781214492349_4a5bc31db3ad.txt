   276→  // ─── Export CSV / Excel ────────────────────────────────────────────
   277→  const flattenData = (data: Record<string, unknown>[], parentKey = ''): Record<string, unknown>[] => {
   278→    return data.map((item) => {
   279→      const flat: Record<string, unknown> = {}
   280→      for (const [key, value] of Object.entries(item)) {
   281→        const newKey = parentKey ? `${parentKey}_${key}` : key
   282→        if (value && typeof value === 'object' && !Array.isArray(value)) {
   283→          Object.assign(flat, flattenData([value as Record<string, unknown>], newKey)[0])
   284→        } else {
   285→          flat[newKey] = value
   286→        }
   287→      }
   288→      return flat
