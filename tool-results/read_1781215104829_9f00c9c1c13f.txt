   234→   234→  const renderPrintContent = () => {
   235→   235→    if (!reportData || !selectedReport) return ''
   236→   236→    const data = reportData as Record<string, unknown>
   237→   237→    const summary = (data.summary || {}) as Record<string, unknown>
   238→   238→    const details = data.details as Record<string, unknown>[] | undefined
   239→   239→    let html = ''
   240→   240→
   241→   241→    // Summary metrics

   244→   244→      html += '<div class="metrics-grid">'
   245→   245→      summaryEntries.slice(0, 8).forEach(([key, value]) => {
   246→   246→        const isCurrency = /revenue|amount|expense|profit|tax|payroll|cost|earning|salary|discount/i.test(key)
   247→   247→        const formatted = typeof value === 'number' ? (isCurrency ? PKR(value) : value.toLocaleString()) : String(value)

   277→   277→  const flattenData = (data: Record<string, unknown>[], parentKey = ''): Record<string, unknown>[] => {
   278→   278→    return data.map((item) => {
   279→   279→      const flat: Record<string, unknown> = {}
   280→   280→      for (const [key, value] of Object.entries(item)) {
   281→   281→        const newKey = parentKey ? `${parentKey}_${key}` : key
   282→   282→        if (value && typeof value === 'object' && !Array.isArray(value)) {
   283→   283→          Object.assign(flat, flattenData([value as Record<string, unknown>], newKey)[0])
   284→   284→        } else {
   285→   285→          flat[newKey] = value
   286→   286→        }
   287→   287→      }
   288→   288→      return flat
   289→   289→    })
