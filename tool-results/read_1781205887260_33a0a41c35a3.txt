    16→// ─── Device Fingerprint Generator ────────────────────────────────────────────
    17→function generateDeviceId(): string {
    18→  if (typeof window === 'undefined') return ''
    19→  const canvas = document.createElement('canvas')
    20→  const ctx = canvas.getContext('2d')
    21→  if (ctx) {
    22→    ctx.textBaseline = 'top'
    23→    ctx.font = '14px Arial'
    24→    ctx.fillText('SalonFlow-Device-Fingerprint', 2, 2)
    25→  }
    26→  const canvasData = canvas.toDataURL()
    27→  const nav = navigator
    28→  const screen = window.screen
    29→  const raw = [
    30→    nav.userAgent,
    31→    nav.language,
    32→    screen.width + 'x' + screen.height,
    33→    screen.colorDepth,
    34→    new Date().getTimezoneOffset(),
    35→    nav.hardwareConcurrency || 'n/a',
    36→    canvasData.slice(-50),
    37→  ].join('||')
    38→  // Simple hash
    39→  let hash = 0
    40→  for (let i = 0; i < raw.length; i++) {
    41→    const char = raw.charCodeAt(i)
    42→    hash = ((hash << 5) - hash) + char
    43→    hash = hash & hash
    44→  }
    45→  return 'SF-DEV-' + Math.abs(hash).toString(36).toUpperCase() + '-' + Date.now().toString(36).toUpperCase()
    46→}

    48→// ─── Live Clock Component ────────────────────────────────────────────────────
    49→function LiveClock() {
    50→  const [time, setTime] = useState(new Date())
    51→
    52→  useEffect(() => {
    53→    const interval = setInterval(() => setTime(new Date()), 1000)
    54→    return () => clearInterval(interval)
    55→  }, [])
    56→
    57→  const timeStr = time.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: true })
    58→  const dateStr = time.toLocaleDateString('en-US', { weekday: 'long', month: 'long', day: 'numeric' })
    59→
    60→  return (
    61→    <motion.div
    62→      initial={{ opacity: 0, y: -20 }}
    63→      animate={{ opacity: 1, y: 0 }}
    64→      transition={{ delay: 0.2, duration: 0.8 }}
    65→      className="text-center mb-8 select-none"
    66→    >
    67→      <p className="text-7xl sm:text-8xl font-extralight text-white tracking-tight drop-shadow-2xl" style={{ fontFeatureSettings: '"tnum"' }}>
    68→        {timeStr}
    69→      </p>
    70→      <p className="text-lg sm:text-xl font-light text-white/70 mt-1 tracking-wide">
    71→        {dateStr}
    72→      </p>
    73→    </motion.div>
    74→  )
    75→}
