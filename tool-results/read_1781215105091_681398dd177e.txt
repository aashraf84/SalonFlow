     5→     5→// ─── Types ────────────────────────────────────────────────────────────────────
     6→     6→
     7→     7→interface ReportResponse {
     8→     8→  success: boolean
     9→     9→  data?: {
    10→    10→    summary: Record<string, unknown>
    11→    11→    charts: Record<string, unknown>
    12→    12→    details: Record<string, unknown>
    13→    13→  }
    14→    14→  error?: string
    15→    15→}
    16→    16→
    17→    17→type ReportType =
    18→    18→  | 'sales-summary'
    19→    19→  | 'sales-item'
    20→    20→  | 'sales-category'
    21→    21→  | 'sales-gross-profit'
    22→    22→  | 'sales-cost'
    23→    23→  | 'sales-retail'
    24→    24→  | 'sales-salesperson'
    25→    25→
    26→    26→// ─── Helpers ──────────────────────────────────────────────────────────────────
    27→    27→
    28→    28→function round2(n: number): number {
    29→    29→  return Math.round(n * 100) / 100
    30→    30→}
    31→    31→
    32→    32→function parseDate(dateStr: string | null, fallback: Date): Date {
    33→    33→  if (!dateStr) return fallback
    34→    34→  const parsed = new Date(dateStr)
    35→    35→  return isNaN(parsed.getTime()) ? fallback : parsed
    36→    36→}
    37→    37→
    38→    38→function buildDateRange(from: string | null, to: string | null) {
    39→    39→  const now = new Date()
    40→    40→  const fromDate = startOfDay(parseDate(from, startOfMonth(now)))
    41→    41→  const toDate = endOfDay(parseDate(to, now))
    42→    42→  return { fromDate, toDate }
    43→    43→}
    44→    44→
    45→    45→function buildInvoiceWhere(fromDate: Date, toDate: Date, branchId?: string | null) {
    46→    46→  const where: Record<string, unknown> = {
    47→    47→    createdAt: { gte: fromDate, lte: toDate },
    48→    48→  }
    49→    49→  if (branchId) {
    50→    50→    where.branchId = branchId
    51→    51→  }
    52→    52→  return where
    53→    53→}

    55→    55→// ─── GET Handler ──────────────────────────────────────────────────────────────
    56→    56→
    57→    57→export async function GET(request: NextRequest): Promise<NextResponse<ReportResponse>> {
    58→    58→  try {
    59→    59→    const { searchParams } = new URL(request.url)
    60→    60→    const type = searchParams.get('type') as ReportType | null
    61→    61→    const from = searchParams.get('from')
    62→    62→    const to = searchParams.get('to')
    63→    63→    const itemType = searchParams.get('itemType') ?? 'all'
    64→    64→    const branchId = searchParams.get('branchId')
    65→    65→    const staffId = searchParams.get('staffId')
    66→    66→
    67→    67→    // Validate report type
    68→    68→    const validTypes: ReportType[] = [
    69→    69→      'sales-summary',
    70→    70→      'sales-item',
    71→    71→      'sales-category',
    72→    72→      'sales-gross-profit',
    73→    73→      'sales-cost',
    74→    74→      'sales-retail',
    75→    75→      'sales-salesperson',
    76→    76→    ]
    77→    77→    if (!type || !validTypes.includes(type)) {
    78→    78→      return NextResponse.json(
    79→    79→        {
    80→    80→          success: false,
    81→    81→          error: `Invalid or missing report type. Valid types: ${validTypes.join(', ')}`,
    82→    82→        },
    83→    83→        { status: 400 }
    84→    84→      )
    85→    85→    }
    86→    86→
    87→    87→    const { fromDate, toDate } = buildDateRange(from, to)
    88→    88→
    89→    89→    switch (type) {
    90→    90→      case 'sales-summary':
    91→    91→        return NextResponse.json({
    92→    92→          success: true,
    93→    93→          data: await getSalesSummary(fromDate, toDate, branchId),
    94→    94→        })
    95→    95→      case 'sales-item':
    96→    96→        return NextResponse.json({
    97→    97→          success: true,
    98→    98→          data: await getSalesItemReport(fromDate, toDate, itemType, branchId),
    99→    99→        })
   100→   100→      case 'sales-category':
   101→   101→        return NextResponse.json({
   102→   102→          success: true,
   103→   103→          data: await getSalesCategoryReport(fromDate, toDate, branchId),
   104→   104→        })
   105→   105→      case 'sales-gross-profit':
   106→   106→        return NextResponse.json({
   107→   107→          success: true,
   108→   108→          data: await getSalesGrossProfitReport(fromDate, toDate, branchId),
   109→   109→        })
   110→   110→      case 'sales-cost':
   111→   111→        return NextResponse.json({
   112→   112→          success: true,
   113→   113→          data: await getSalesCostReport(fromDate, toDate, branchId),
   114→   114→        })
   115→   115→      case 'sales-retail':
   116→   116→        return NextResponse.json({
   117→   117→          success: true,
   118→   118→          data: await getSalesRetailReport(fromDate, toDate, branchId),
   119→   119→        })
   120→   120→      case 'sales-salesperson':
   121→   121→        return NextResponse.json({
   122→   122→          success: true,
   123→   123→          data: await getSalesSalespersonReport(fromDate, toDate, branchId, staffId),
   124→   124→        })
   125→   125→      default:
   126→   126→        return NextResponse.json({ success: false, error: 'Unhandled report type' }, { status: 400 })
   127→   127→    }
   128→   128→  } catch (error: unknown) {
   129→   129→    const message = error instanceof Error ? error.message : 'Unknown error occurred'
   130→   130→    console.error('[Sales Reports API]', message)
   131→   131→    return NextResponse.json({ success: false, error: message }, { status: 500 })
   132→   132→  }
   133→   133→}

   135→   135→// ─── 1. Sales Summary ─────────────────────────────────────────────────────────
   136→   136→
   137→   137→async function getSalesSummary(fromDate: Date, toDate: Date, branchId?: string | null) {
   138→   138→  const invoiceWhere = buildInvoiceWhere(fromDate, toDate, branchId)
   139→   139→
   140→   140→  // Fetch all invoices with items in the date range
   141→   141→  const invoices = await db.invoice.findMany({
   142→   142→    where: invoiceWhere,
   143→   143→    include: {
   144→   144→      items: true,
   145→   145→      branch: { select: { id: true, name: true } },
   146→   146→    },
   147→   147→  })
   148→   148→
   149→   149→  // Separate regular sales vs refunds
   150→   150→  const paidInvoices = invoices.filter((inv) => inv.paymentStatus !== 'refunded')
   151→   151→  const refundedInvoices = invoices.filter((inv) => inv.paymentStatus === 'refunded')
   152→   152→
   153→   153→  const totalSales = round2(paidInvoices.reduce((sum, inv) => sum + inv.totalAmount, 0))
   154→   154→  const totalReturns = round2(refundedInvoices.reduce((sum, inv) => sum + inv.totalAmount, 0))
   155→   155→  const netSales = round2(totalSales - totalReturns)
   156→   156→  const totalTax = round2(paidInvoices.reduce((sum, inv) => sum + inv.tax, 0))
   157→   157→  const totalDiscount = round2(paidInvoices.reduce((sum, inv) => sum + inv.discount, 0))
   158→   158→
   159→   159→  // Calculate COGS for product items in paid invoices
   160→   160→  const productItems = paidInvoices.flatMap((inv) =>
   161→   161→    inv.items.filter((item) => item.itemType === 'product')
   162→   162→  )
   163→   163→  const productCostMap = await buildProductCostMap(productItems.map((i) => i.itemId).filter(Boolean) as string[])
   164→   164→  const totalCOGS = round2(
   165→   165→    productItems.reduce((sum, item) => {
   166→   166→      const costPrice = productCostMap[item.itemId ?? ''] ?? 0
   167→   167→      return sum + costPrice * item.quantity
   168→   168→    }, 0)
   169→   169→  )
   170→   170→
   171→   171→  const grossProfit = round2(netSales - totalCOGS)
   172→   172→  const grossProfitMargin = netSales > 0 ? round2((grossProfit / netSales) * 100) : 0
   173→   173→
   174→   174→  // Average order value
   175→   175→  const totalPaidInvoices = paidInvoices.length
   176→   176→  const avgOrderValue = totalPaidInvoices > 0 ? round2(totalSales / totalPaidInvoices) : 0
   177→   177→
   178→   178→  // Total items sold
   179→   179→  const allPaidItems = paidInvoices.flatMap((inv) => inv.items)
   180→   180→  const totalItemsSold = allPaidItems.reduce((sum, item) => sum + item.quantity, 0)
   181→   181→
   182→   182→  // Sales by invoice type
   183→   183→  const salesByType: Record<string, { count: number; revenue: number }> = {}
   184→   184→  for (const inv of paidInvoices) {
   185→   185→    if (!salesByType[inv.type]) {
   186→   186→      salesByType[inv.type] = { count: 0, revenue: 0 }
   187→   187→    }
   188→   188→    salesByType[inv.type].count++
   189→   189→    salesByType[inv.type].revenue += inv.totalAmount
   190→   190→  }
   191→   191→  for (const key of Object.keys(salesByType)) {
   192→   192→    salesByType[key].revenue = round2(salesByType[key].revenue)
   193→   193→  }
   194→   194→
   195→   195→  // Payment method breakdown
   196→   196→  const paymentMethodBreakdown: Record<string, { count: number; revenue: number }> = {}
   197→   197→  for (const inv of paidInvoices) {
   198→   198→    if (!paymentMethodBreakdown[inv.paymentMethod]) {
   199→   199→      paymentMethodBreakdown[inv.paymentMethod] = { count: 0, revenue: 0 }
   200→   200→    }
   201→   201→    paymentMethodBreakdown[inv.paymentMethod].count++
   202→   202→    paymentMethodBreakdown[inv.paymentMethod].revenue += inv.totalAmount
   203→   203→  }
   204→   204→  for (const key of Object.keys(paymentMethodBreakdown)) {
   205→   205→    paymentMethodBreakdown[key].revenue = round2(paymentMethodBreakdown[key].revenue)
   206→   206→  }
   207→   207→
   208→   208→  // Daily sales trend
   209→   209→  const dailyTrend: Record<string, { sales: number; invoices: number }> = {}
   210→   210→  for (const inv of paidInvoices) {
   211→   211→    const dayKey = format(new Date(inv.createdAt), 'yyyy-MM-dd')
   212→   212→    if (!dailyTrend[dayKey]) {
   213→   213→      dailyTrend[dayKey] = { sales: 0, invoices: 0 }
   214→   214→    }
   215→   215→    dailyTrend[dayKey].sales += inv.totalAmount
   216→   216→    dailyTrend[dayKey].invoices++
   217→   217→  }
   218→   218→  for (const key of Object.keys(dailyTrend)) {
   219→   219→    dailyTrend[key].sales = round2(dailyTrend[key].sales)
   220→   220→  }
   221→   221→
   222→   222→  // Monthly sales trend
   223→   223→  const monthlyTrend: Record<string, { sales: number; invoices: number; refunds: number }> = {}
   224→   224→  for (const inv of paidInvoices) {
   225→   225→    const monthKey = format(new Date(inv.createdAt), 'yyyy-MM')
   226→   226→    if (!monthlyTrend[monthKey]) {
   227→   227→      monthlyTrend[monthKey] = { sales: 0, invoices: 0, refunds: 0 }
   228→   228→    }
   229→   229→    monthlyTrend[monthKey].sales += inv.totalAmount
   230→   230→    monthlyTrend[monthKey].invoices++
   231→   231→  }
   232→   232→  for (const inv of refundedInvoices) {
   233→   233→    const monthKey = format(new Date(inv.createdAt), 'yyyy-MM')
   234→   234→    if (!monthlyTrend[monthKey]) {
   235→   235→      monthlyTrend[monthKey] = { sales: 0, invoices: 0, refunds: 0 }
   236→   236→    }
   237→   237→    monthlyTrend[monthKey].refunds += inv.totalAmount
   238→   238→  }
   239→   239→  for (const key of Object.keys(monthlyTrend)) {
   240→   240→    monthlyTrend[key].sales = round2(monthlyTrend[key].sales)
   241→   241→    monthlyTrend[key].refunds = round2(monthlyTrend[key].refunds)
   242→   242→  }
   243→   243→
   244→   244→  // Top 10 selling items (by revenue)
   245→   245→  const itemRevenueMap: Record<string, { name: string; type: string; qty: number; revenue: number }> = {}
   246→   246→  for (const item of allPaidItems) {
   247→   247→    const key = `${item.name}|||${item.itemType}`
   248→   248→    if (!itemRevenueMap[key]) {
   249→   249→      itemRevenueMap[key] = { name: item.name, type: item.itemType, qty: 0, revenue: 0 }
   250→   250→    }
   251→   251→    itemRevenueMap[key].qty += item.quantity
   252→   252→    itemRevenueMap[key].revenue += item.total
   253→   253→  }
   254→   254→  const top10Items = Object.values(itemRevenueMap)
   255→   255→    .sort((a, b) => b.revenue - a.revenue)
   256→   256→    .slice(0, 10)
   257→   257→    .map((item) => ({ ...item, revenue: round2(item.revenue) }))
   258→   258→
   259→   259→  return {
   260→   260→    summary: {
   261→   261→      totalSales,
   262→   262→      totalReturns,
   263→   263→      netSales,
   264→   264→      totalTax,
   265→   265→      totalDiscount,
   266→   266→      totalCOGS,
   267→   267→      grossProfit,
   268→   268→      grossProfitMargin,
   269→   269→      avgOrderValue,
   270→   270→      totalItemsSold,
   271→   271→      totalInvoices: invoices.length,
   272→   272→      paidInvoices: totalPaidInvoices,
   273→   273→      refundedInvoices: refundedInvoices.length,
   274→   274→      salesByType,
   275→   275→      paymentMethodBreakdown,
   276→   276→    },
   277→   277→    charts: {
   278→   278→      dailyTrend,
   279→   279→      monthlyTrend,
   280→   280→      top10Items,
   281→   281→      salesByTypeDistribution: Object.entries(salesByType).map(([type, data]) => ({
   282→   282→        type,
   283→   283→        ...data,
   284→   284→        percentage: totalSales > 0 ? round2((data.revenue / totalSales) * 100) : 0,
   285→   285→      })),
   286→   286→      paymentMethodDistribution: Object.entries(paymentMethodBreakdown).map(([method, data]) => ({
   287→   287→        method,
   288→   288→        ...data,
   289→   289→        percentage: totalSales > 0 ? round2((data.revenue / totalSales) * 100) : 0,
   290→   290→      })),
   291→   291→    },
   292→   292→    details: {
   293→   293→      dateRange: {
   294→   294→        from: format(fromDate, 'yyyy-MM-dd'),
   295→   295→        to: format(toDate, 'yyyy-MM-dd'),
   296→   296→      },
   297→   297→      branchId: branchId || null,
   298→   298→    },
   299→   299→  }
   300→   300→}

   302→   302→// ─── 2. Item-wise Sales Report ────────────────────────────────────────────────
   303→   303→
   304→   304→async function getSalesItemReport(
   305→   305→  fromDate: Date,
   306→   306→  toDate: Date,
   307→   307→  itemType: string,
   308→   308→  branchId?: string | null
   309→   309→) {
   310→   310→  const invoiceWhere = buildInvoiceWhere(fromDate, toDate, branchId)
   311→   311→
   312→   312→  const invoices = await db.invoice.findMany({
   313→   313→    where: { ...invoiceWhere, paymentStatus: { not: 'refunded' } },
   314→   314→    include: { items: true },
   315→   315→  })
   316→   316→
   317→   317→  const allItems = invoices.flatMap((inv) => inv.items)
   318→   318→
   319→   319→  // Filter by itemType if specified
   320→   320→  const filteredItems =
   321→   321→    itemType === 'all' ? allItems : allItems.filter((item) => item.itemType === itemType)
   322→   322→
   323→   323→  // Group by name + itemType
   324→   324→  const itemMap: Record<
   325→   325→    string,
   326→   326→    { name: string; itemType: string; qty: number; totalRevenue: number; totalDiscount: number; count: number }
   327→   327→  > = {}
   328→   328→  for (const item of filteredItems) {
   329→   329→    const key = `${item.name}|||${item.itemType}`
   330→   330→    if (!itemMap[key]) {
   331→   331→      itemMap[key] = { name: item.name, itemType: item.itemType, qty: 0, totalRevenue: 0, totalDiscount: 0, count: 0 }
   332→   332→    }
   333→   333→    itemMap[key].qty += item.quantity
   334→   334→    itemMap[key].totalRevenue += item.total
   335→   335→    itemMap[key].totalDiscount += item.discount
   336→   336→    itemMap[key].count++
   337→   337→  }
   338→   338→
   339→   339→  const totalSales = round2(filteredItems.reduce((sum, item) => sum + item.total, 0))
   340→   340→
   341→   341→  // Enrich with category info
   342→   342→  const serviceCategoryMap = await buildServiceCategoryMap()
   343→   343→  const productCategoryMap = await buildProductCategoryMap()
   344→   344→
   345→   345→  const items = Object.values(itemMap).map((item) => {
   346→   346→    const avgDiscount = item.count > 0 ? round2(item.totalDiscount / item.count) : 0
   347→   347→    const percentageOfTotal = totalSales > 0 ? round2((item.totalRevenue / totalSales) * 100) : 0
   348→   348→    const unitPrice = item.qty > 0 ? round2(item.totalRevenue / item.qty) : 0
   349→   349→
   350→   350→    const category =
   351→   351→      item.itemType === 'service'
   352→   352→        ? (serviceCategoryMap[item.name] ?? 'Uncategorized')
   353→   353→        : item.itemType === 'product'
   354→   354→          ? (productCategoryMap[item.name] ?? 'Uncategorized')
   355→   355→          : 'N/A'
   356→   356→
   357→   357→    return {
   358→   358→      name: item.name,
   359→   359→      type: item.itemType,
   360→   360→      category,
   361→   361→      qtySold: item.qty,
   362→   362→      unitPrice,
   363→   363→      totalRevenue: round2(item.totalRevenue),
   364→   364→      percentageOfTotal,
   365→   365→      avgDiscount,
   366→   366→    }
   367→   367→  })
   368→   368→
   369→   369→  // Sort by revenue (default)
   370→   370→  items.sort((a, b) => b.totalRevenue - a.totalRevenue)
   371→   371→
   372→   372→  // Top 20 for bar chart
   373→   373→  const top20Items = items.slice(0, 20)
   374→   374→
   375→   375→  // Top and bottom performers
   376→   376→  const topPerformers = items.slice(0, 5)
   377→   377→  const bottomPerformers = items.length > 5 ? items.slice(-5).reverse() : []
   378→   378→
   379→   379→  // Also sort by quantity
   380→   380→  const itemsByQty = [...items].sort((a, b) => b.qtySold - a.qtySold)
   381→   381→
   382→   382→  return {
   383→   383→    summary: {
   384→   384→      totalItems: items.length,
   385→   385→      totalQtySold: items.reduce((sum, i) => sum + i.qtySold, 0),
   386→   386→      totalRevenue: round2(items.reduce((sum, i) => sum + i.totalRevenue, 0)),
   387→   387→      avgItemRevenue: items.length > 0 ? round2(items.reduce((sum, i) => sum + i.totalRevenue, 0) / items.length) : 0,
   388→   388→      totalDiscount: round2(filteredItems.reduce((sum, i) => sum + i.discount, 0)),
   389→   389→      filteredBy: itemType,
   390→   390→    },
   391→   391→    charts: {
   392→   392→      top20Items: top20Items.map((item) => ({
   393→   393→        name: item.name.length > 20 ? item.name.substring(0, 20) + '...' : item.name,
   394→   394→        revenue: item.totalRevenue,
   395→   395→        qty: item.qtySold,
   396→   396→        type: item.type,
   397→   397→      })),
   398→   398→      revenueByType: Object.entries(
   399→   399→        items.reduce<Record<string, number>>((acc, i) => {
   400→   400→          acc[i.type] = (acc[i.type] || 0) + i.totalRevenue
   401→   401→          return acc
   402→   402→        }, {})
   403→   403→      ).map(([type, revenue]) => ({ type, revenue: round2(revenue) })),
   404→   404→    },
   405→   405→    details: {
   406→   406→      items,
   407→   407→      itemsByQty,
   408→   408→      topPerformers,
   409→   409→      bottomPerformers,
   410→   410→    },
   411→   411→  }
   412→   412→}

   414→   414→// ─── 3. Category-wise Sales Report ────────────────────────────────────────────
   415→   415→
   416→   416→async function getSalesCategoryReport(fromDate: Date, toDate: Date, branchId?: string | null) {
   417→   417→  const invoiceWhere = buildInvoiceWhere(fromDate, toDate, branchId)
   418→   418→
   419→   419→  const invoices = await db.invoice.findMany({
   420→   420→    where: { ...invoiceWhere, paymentStatus: { not: 'refunded' } },
   421→   421→    include: { items: true },
   422→   422→  })
   423→   423→
   424→   424→  const allItems = invoices.flatMap((inv) => inv.items)
   425→   425→
   426→   426→  // Fetch all service and product categories with their items
   427→   427→  const serviceCategories = await db.serviceCategory.findMany({
   428→   428→    include: { services: { select: { id: true, name: true, price: true } } },
   429→   429→  })
   430→   430→
   431→   431→  const productCategories = await db.productCategory.findMany({
   432→   432→    include: { products: { select: { id: true, name: true, sellingPrice: true, costPrice: true } } },
   433→   433→  })
   434→   434→
   435→   435→  // Build lookup maps: itemName -> category info
   436→   436→  const serviceToCategory: Record<string, { categoryId: string; categoryName: string; price: number }> = {}
   437→   437→  for (const cat of serviceCategories) {
   438→   438→    for (const svc of cat.services) {
   439→   439→      serviceToCategory[svc.name] = { categoryId: cat.id, categoryName: cat.name, price: svc.price }
   440→   440→    }
   441→   441→  }
   442→   442→
   443→   443→  const productToCategory: Record<string, { categoryId: string; categoryName: string; sellingPrice: number; costPrice: number }> = {}
   444→   444→  for (const cat of productCategories) {
   445→   445→    for (const prod of cat.products) {
   446→   446→      productToCategory[prod.name] = { categoryId: cat.id, categoryName: cat.name, sellingPrice: prod.sellingPrice, costPrice: prod.costPrice }
   447→   447→    }
   448→   448→  }
   449→   449→
   450→   450→  // Aggregate service items by category
   451→   451→  const serviceCategoryAgg: Record<string, { categoryName: string; totalSold: number; revenue: number; totalPrice: number; count: number }> = {}
   452→   452→  const serviceItems = allItems.filter((item) => item.itemType === 'service')
   453→   453→
   454→   454→  for (const item of serviceItems) {
   455→   455→    const catInfo = serviceToCategory[item.name]
   456→   456→    const catName = catInfo?.categoryName ?? 'Uncategorized'
   457→   457→    if (!serviceCategoryAgg[catName]) {
   458→   458→      serviceCategoryAgg[catName] = { categoryName: catName, totalSold: 0, revenue: 0, totalPrice: 0, count: 0 }
   459→   459→    }
   460→   460→    serviceCategoryAgg[catName].totalSold += item.quantity
   461→   461→    serviceCategoryAgg[catName].revenue += item.total
   462→   462→    serviceCategoryAgg[catName].totalPrice += item.unitPrice
   463→   463→    serviceCategoryAgg[catName].count++
   464→   464→  }
   465→   465→
   466→   466→  const totalServiceRevenue = round2(serviceItems.reduce((sum, i) => sum + i.total, 0))
   467→   467→
   468→   468→  const serviceCategoryReport = Object.values(serviceCategoryAgg).map((cat) => ({
   469→   469→    categoryName: cat.categoryName,
   470→   470→    totalSold: cat.totalSold,
   471→   471→    revenue: round2(cat.revenue),
   472→   472→    avgPrice: cat.count > 0 ? round2(cat.totalPrice / cat.count) : 0,
   473→   473→    percentageOfServiceRevenue: totalServiceRevenue > 0 ? round2((cat.revenue / totalServiceRevenue) * 100) : 0,
   474→   474→  }))
   475→   475→
   476→   476→  // Aggregate product items by category
   477→   477→  const productCategoryAgg: Record<string, { categoryName: string; totalSold: number; revenue: number; totalSellingPrice: number; totalCostPrice: number; count: number }> = {}
   478→   478→  const productItems = allItems.filter((item) => item.itemType === 'product')
   479→   479→
   480→   480→  for (const item of productItems) {
   481→   481→    const catInfo = productToCategory[item.name]
   482→   482→    const catName = catInfo?.categoryName ?? 'Uncategorized'
   483→   483→    if (!productCategoryAgg[catName]) {
   484→   484→      productCategoryAgg[catName] = { categoryName: catName, totalSold: 0, revenue: 0, totalSellingPrice: 0, totalCostPrice: 0, count: 0 }
   485→   485→    }
   486→   486→    productCategoryAgg[catName].totalSold += item.quantity
   487→   487→    productCategoryAgg[catName].revenue += item.total
   488→   488→    productCategoryAgg[catName].totalSellingPrice += item.unitPrice
   489→   489→    productCategoryAgg[catName].totalCostPrice += catInfo?.costPrice ?? 0
   490→   490→    productCategoryAgg[catName].count++
   491→   491→  }
   492→   492→
   493→   493→  const totalProductRevenue = round2(productItems.reduce((sum, i) => sum + i.total, 0))

   527→   527→      }
   528→   528→    })
   529→   529→    .sort((a, b) => b.totalRevenue - a.totalRevenue)
   530→   530→
   531→   531→  // Pie chart data
   532→   532→  const pieData = [
   533→   533→    ...serviceCategoryReport.map((c) => ({ name: `${c.categoryName} (Service)`, value: c.revenue, type: 'service' })),
   534→   534→    ...productCategoryReport.map((c) => ({ name: `${c.categoryName} (Product)`, value: c.revenue, type: 'product' })),

   540→   540→      totalProductCategories: productCategoryReport.length,
   541→   541→      totalServiceRevenue,
   542→   542→      totalProductRevenue,
   543→   543→      combinedRevenue: round2(totalServiceRevenue + totalProductRevenue),
   544→   544→    },
   545→   545→    charts: {
   546→   546→      serviceCategoryDistribution: serviceCategoryReport.map((c) => ({
   547→   547→        name: c.categoryName,
   548→   548→        value: c.revenue,
   549→   549→        percentage: c.percentageOfServiceRevenue,

   551→   551→      productCategoryDistribution: productCategoryReport.map((c) => ({

   553→   553→        value: c.revenue,

   556→   556→      pieData,

   566→   566→      productCategories: productCategoryReport.sort((a, b) => b.revenue - a.revenue),
   567→   567→      combinedView,
   568→   568→    },
   569→   569→  }

   574→   574→async function getSalesGrossProfitReport(fromDate: Date, toDate: Date, branchId?: string | null) {
   575→   575→  const invoiceWhere = buildInvoiceWhere(fromDate, toDate, branchId)
   576→   576→
   577→   577→  const invoices = await db.invoice.findMany({
   578→   578→    where: { ...invoiceWhere, paymentStatus: { not: 'refunded' } },
   579→   579→    include: { items: true },
   580→   580→  })
   581→   581→
   582→   582→  const allItems = invoices.flatMap((inv) => inv.items)
   583→   583→
   584→   584→  // Revenue by type
   585→   585→  const serviceRevenue = round2(allItems.filter((i) => i.itemType === 'service').reduce((sum, i) => sum + i.total, 0))
   586→   586→  const productRevenue = round2(allItems.filter((i) => i.itemType === 'product').reduce((sum, i) => sum + i.total, 0))
   587→   587→  const membershipRevenue = round2(allItems.filter((i) => i.itemType === 'membership').reduce((sum, i) => sum + i.total, 0))
   588→   588→  const packageRevenue = round2(allItems.filter((i) => i.itemType === 'package').reduce((sum, i) => sum + i.total, 0))
   589→   589→
   590→   590→  // COGS - only for product items
   591→   591→  const productItems = allItems.filter((i) => i.itemType === 'product')
   592→   592→  const productCostMap = await buildProductCostMap(productItems.map((i) => i.itemId).filter(Boolean) as string[])
   593→   593→
   594→   594→  const totalCOGS = round2(
   595→   595→    productItems.reduce((sum, item) => {
   596→   596→      const costPrice = productCostMap[item.itemId ?? ''] ?? 0
   597→   597→      return sum + costPrice * item.quantity
   598→   598→    }, 0)
   599→   599→  )
   600→   600→
   601→   601→  // Gross profit: Services have no COGS, so GP = Revenue for services
   602→   602→  const serviceGP = round2(serviceRevenue) // No COGS for services
   603→   603→  const productGP = round2(productRevenue - totalCOGS)
   604→   604→  const totalRevenue = round2(serviceRevenue + productRevenue + membershipRevenue + packageRevenue)
   605→   605→  const totalGP = round2(serviceGP + productGP)
   606→   606→  const totalGPMargin = totalRevenue > 0 ? round2((totalGP / totalRevenue) * 100) : 0
   607→   607→
   608→   608→  const serviceGPMargin = serviceRevenue > 0 ? round2((serviceGP / serviceRevenue) * 100) : 0
   609→   609→  const productGPMargin = productRevenue > 0 ? round2((productGP / productRevenue) * 100) : 0
   610→   610→
   611→   611→  // Monthly gross profit trend
   612→   612→  const monthlyGP: Record<string, { revenue: number; cogs: number; grossProfit: number; margin: number }> = {}
   613→   613→
   614→   614→  for (const inv of invoices) {
   615→   615→    const monthKey = format(new Date(inv.createdAt), 'yyyy-MM')
   616→   616→    if (!monthlyGP[monthKey]) {
   617→   617→      monthlyGP[monthKey] = { revenue: 0, cogs: 0, grossProfit: 0, margin: 0 }
   618→   618→    }
   619→   619→    monthlyGP[monthKey].revenue += inv.totalAmount
   620→   620→  }
   621→   621→
   622→   622→  // Calculate monthly COGS
   623→   623→  for (const inv of invoices) {
   624→   624→    const monthKey = format(new Date(inv.createdAt), 'yyyy-MM')
   625→   625→    const invProductItems = inv.items.filter((i) => i.itemType === 'product')
   626→   626→    for (const item of invProductItems) {
   627→   627→      const costPrice = productCostMap[item.itemId ?? ''] ?? 0
   628→   628→      monthlyGP[monthKey].cogs += costPrice * item.quantity
   629→   629→    }
   630→   630→  }
   631→   631→
   632→   632→  for (const key of Object.keys(monthlyGP)) {
   633→   633→    monthlyGP[key].revenue = round2(monthlyGP[key].revenue)
   634→   634→    monthlyGP[key].cogs = round2(monthlyGP[key].cogs)
   635→   635→    monthlyGP[key].grossProfit = round2(monthlyGP[key].revenue - monthlyGP[key].cogs)
   636→   636→    monthlyGP[key].margin = monthlyGP[key].revenue > 0 ? round2((monthlyGP[key].grossProfit / monthlyGP[key].revenue) * 100) : 0
   637→   637→  }
   638→   638→
   639→   639→  // Product-level gross profit breakdown
   640→   640→  const productGPAgg: Record<string, { name: string; revenue: number; cost: number; qty: number }> = {}
   641→   641→  for (const item of productItems) {
   642→   642→    if (!productGPAgg[item.name]) {
   643→   643→      productGPAgg[item.name] = { name: item.name, revenue: 0, cost: 0, qty: 0 }
   644→   644→    }
   645→   645→    productGPAgg[item.name].revenue += item.total
   646→   646→    productGPAgg[item.name].cost += (productCostMap[item.itemId ?? ''] ?? 0) * item.quantity
   647→   647→    productGPAgg[item.name].qty += item.quantity
   648→   648→  }
   649→   649→
   650→   650→  const productGPBreakdown = Object.values(productGPAgg)
   651→   651→    .map((p) => {
   652→   652→      const gp = round2(p.revenue - p.cost)
   653→   653→      return {
   654→   654→        name: p.name,
   655→   655→        revenue: round2(p.revenue),
   656→   656→        cost: round2(p.cost),
   657→   657→        grossProfit: gp,
   658→   658→        margin: p.revenue > 0 ? round2((gp / p.revenue) * 100) : 0,
   659→   659→      }
   660→   660→    })
   661→   661→    .sort((a, b) => b.grossProfit - a.grossProfit)
   662→   662→
   663→   663→  // Service-level contribution
   664→   664→  const serviceGPAgg: Record<string, { name: string; revenue: number; qty: number }> = {}
   665→   665→  const serviceItems = allItems.filter((i) => i.itemType === 'service')
   666→   666→  for (const item of serviceItems) {
   667→   667→    if (!serviceGPAgg[item.name]) {
   668→   668→      serviceGPAgg[item.name] = { name: item.name, revenue: 0, qty: 0 }
   669→   669→    }
   670→   670→    serviceGPAgg[item.name].revenue += item.total
   671→   671→    serviceGPAgg[item.name].qty += item.quantity
   672→   672→  }
   673→   673→
   674→   674→  const serviceGPBreakdown = Object.values(serviceGPAgg)
   675→   675→    .map((s) => ({
   676→   676→      name: s.name,
   677→   677→      revenue: round2(s.revenue),
   678→   678→      grossProfit: round2(s.revenue), // Pure margin - no COGS
   679→   679→      margin: 100, // Services have 100% gross margin (no COGS)
   680→   680→      qty: s.qty,
   681→   681→    }))
   682→   682→    .sort((a, b) => b.grossProfit - a.grossProfit)
   683→   683→
   684→   684→  return {
   685→   685→    summary: {
   686→   686→      totalRevenue,
   687→   687→      serviceRevenue,
   688→   688→      productRevenue,
   689→   689→      membershipRevenue,
   690→   690→      packageRevenue,
   691→   691→      totalCOGS,
   692→   692→      totalGP,
   693→   693→      totalGPMargin,
   694→   694→      serviceGP,
   695→   695→      serviceGPMargin,
   696→   696→      productGP,
   697→   697→      productGPMargin,
   698→   698→    },
   699→   699→    charts: {
   700→   700→      revenueByType: [
   701→   701→        { type: 'Service', revenue: serviceRevenue, gp: serviceGP, margin: serviceGPMargin },
   702→   702→        { type: 'Product', revenue: productRevenue, gp: productGP, margin: productGPMargin },
   703→   703→        ...(membershipRevenue > 0 ? [{ type: 'Membership', revenue: membershipRevenue, gp: round2(membershipRevenue), margin: 100 }] : []),
   704→   704→        ...(packageRevenue > 0 ? [{ type: 'Package', revenue: packageRevenue, gp: round2(packageRevenue), margin: 100 }] : []),
   705→   705→      ],
   706→   706→      monthlyGPTrend: Object.entries(monthlyGP)
   707→   707→        .sort(([a], [b]) => a.localeCompare(b))
   708→   708→        .map(([month, data]) => ({ month, ...data })),
   709→   709→    },
   710→   710→    details: {
   711→   711→      productGPBreakdown,
   712→   712→      serviceGPBreakdown,
   713→   713→      topProfitableProducts: productGPBreakdown.slice(0, 10),
   714→   714→      topProfitableServices: serviceGPBreakdown.slice(0, 10),
   715→   715→    },
   716→   716→  }
   717→   717→}

   830→   830→

   859→   859→  const totalFixedCosts = round2(fixedExpenseResult._sum.amount ?? 0)

   861→   861→  const totalVariableCost = round2(
   862→   862→    Object.values(cogsByCategory).reduce((sum, c) => sum + c.totalCost, 0)

   865→   865→    Object.values(cogsByCategory).reduce((sum, c) => sum + c.totalRevenue, 0)
   866→   866→  )
   867→   867→  const contributionMargin = totalProductRevenue > 0 ? round2(((totalProductRevenue - totalVariableCost) / totalProductRevenue) * 100) : 0
   868→   868→  const breakEvenRevenue = contributionMargin > 0 ? round2((totalFixedCosts / contributionMargin) * 100) : 0
   869→   869→
   870→   870→  return {
   871→   871→    summary: {

   873→   873→      totalProductRevenue,
   874→   874→      totalFixedCosts,
   875→   875→      totalVariableCost,
   876→   876→      contributionMargin,
   877→   877→      breakEvenRevenue,
   878→   878→    },
   879→   879→    charts: {
   880→   880→      cogsByCategory: cogsByCategoryReport,
   881→   881→      monthlyCostTrend: Object.entries(monthlyCost)
   882→   882→        .sort(([a], [b]) => a.localeCompare(b))
   883→   883→        .map(([month, data]) => ({ month, ...data })),
   884→   884→      markupByCategory: cogsByCategoryReport.map((c) => ({
   885→   885→        categoryName: c.categoryName,
   886→   886→        markup: c.markup,
   887→   887→        totalCost: c.totalCost,
   888→   888→        totalRevenue: c.totalRevenue,
   889→   889→      })),
   890→   890→    },
   891→   891→    details: {
   892→   892→      costVsPrice: costVsPriceReport,
   893→   893→      highestMargin,
   894→   894→      lowestMargin,
   895→   895→    },
   896→   896→  }

   901→   901→async function getSalesRetailReport(fromDate: Date, toDate: Date, branchId?: string | null) {

   936→   936→    const costPrice = detail?.costPrice ?? 0
   937→   937→    const totalCost = costPrice * item.quantity
   938→   938→    const grossMargin = round2(item.total - totalCost)
   939→   939→    const marginPercent = item.total > 0 ? round2((grossMargin / item.total) * 100) : 0
   940→   940→

   943→   943→      sku: detail?.sku ?? 'N/A',

   949→   949→      costPrice: round2(costPrice),
   950→   950→      totalCost: round2(totalCost),
   951→   951→      grossMargin,
   952→   952→      marginPercent,
   953→   953→    }
   954→   954→  })
   955→   955→
   956→   956→  // Aggregate by product name (unique products)
   957→   957→  const productAgg: Record<string, {
   958→   958→    name: string; sku: string; brand: string; category: string;
   959→   959→    qtySold: number; totalRevenue: number; totalCost: number

   961→   961→  for (const item of retailItems) {

   964→   964→    }

   968→   968→  }
   969→   969→

   971→   971→    .map((p) => {

   974→   974→        ...p,
   975→   975→        totalRevenue: round2(p.totalRevenue),
   976→   976→        totalCost: round2(p.totalCost),
   977→   977→        grossMargin: gm,
   978→   978→        marginPercent: p.totalRevenue > 0 ? round2((gm / p.totalRevenue) * 100) : 0,
   979→   979→      }
   980→   980→    })
   981→   981→    .sort((a, b) => b.totalRevenue - a.totalRevenue)
   982→   982→
   983→   983→  // Sales by brand
   984→   984→  const brandAgg: Record<string, { brand: string; qtySold: number; revenue: number }> = {}
   985→   985→  for (const item of retailItems) {
   986→   986→    const brand = item.brand || 'Unbranded'
   987→   987→    if (!brandAgg[brand]) {
   988→   988→      brandAgg[brand] = { brand, qtySold: 0, revenue: 0 }
   989→   989→    }
   990→   990→    brandAgg[brand].qtySold += item.qtySold
   991→   991→    brandAgg[brand].revenue += item.totalRevenue
   992→   992→  }
   993→   993→  const salesByBrand = Object.values(brandAgg)
   994→   994→    .map((b) => ({ ...b, revenue: round2(b.revenue) }))
   995→   995→    .sort((a, b) => b.revenue - a.revenue)
   996→   996→
   997→   997→  // Sales by product category
   998→   998→  const catAgg: Record<string, { category: string; qtySold: number; revenue: number }> = {}
   999→   999→  for (const item of retailItems) {
  1000→  1000→    if (!catAgg[item.category]) {
  1001→  1001→      catAgg[item.category] = { category: item.category, qtySold: 0, revenue: 0 }
  1002→  1002→    }
  1003→  1003→    catAgg[item.category].qtySold += item.qtySold
  1004→  1004→    catAgg[item.category].revenue += item.totalRevenue
  1005→  1005→  }
  1006→  1006→  const salesByCategory = Object.values(catAgg)
  1007→  1007→    .map((c) => ({ ...c, revenue: round2(c.revenue) }))
  1008→  1008→    .sort((a, b) => b.revenue - a.revenue)
  1009→  1009→
  1010→  1010→  // Top retail performers (by revenue)
  1011→  1011→  const topPerformers = productSummary.slice(0, 10)
  1012→  1012→
  1013→  1013→  // Slow-moving items (low qty sold)
  1014→  1014→  const slowMoving = [...productSummary].sort((a, b) => a.qtySold - b.qtySold).slice(0, 10)
  1015→  1015→
  1016→  1016→  // Retail vs Service comparison
  1017→  1017→  const retailRevenue = round2(productItems.reduce((sum, i) => sum + i.total, 0))
  1018→  1018→  const serviceRevenue = round2(serviceItems.reduce((sum, i) => sum + i.total, 0))
  1019→  1019→  const combinedRevenue = round2(retailRevenue + serviceRevenue)
  1020→  1020→
  1021→  1021→  // Monthly retail trend
  1022→  1022→  const monthlyRetail: Record<string, { retailRevenue: number; retailQty: number; serviceRevenue: number }> = {}

  1025→  1025→    if (!monthlyRetail[monthKey]) {
  1026→  1026→      monthlyRetail[monthKey] = { retailRevenue: 0, retailQty: 0, serviceRevenue: 0 }
  1027→  1027→    }
  1028→  1028→    for (const item of inv.items) {
  1029→  1029→      if (item.itemType === 'product') {
  1030→  1030→        monthlyRetail[monthKey].retailRevenue += item.total
  1031→  1031→        monthlyRetail[monthKey].retailQty += item.quantity
  1032→  1032→      } else if (item.itemType === 'service') {
  1033→  1033→        monthlyRetail[monthKey].serviceRevenue += item.total
  1034→  1034→      }

  1036→  1036→  }
  1037→  1037→  for (const key of Object.keys(monthlyRetail)) {
  1038→  1038→    monthlyRetail[key].retailRevenue = round2(monthlyRetail[key].retailRevenue)
  1039→  1039→    monthlyRetail[key].serviceRevenue = round2(monthlyRetail[key].serviceRevenue)
  1040→  1040→  }
  1041→  1041→
  1042→  1042→  return {

  1049→  1049→      uniqueProducts: productSummary.length,
  1050→  1050→      avgRetailMargin: productSummary.length > 0 ? round2(productSummary.reduce((sum, p) => sum + p.marginPercent, 0) / productSummary.length) : 0,
  1051→  1051→    },
  1052→  1052→    charts: {
  1053→  1053→      salesByBrand,
  1054→  1054→      salesByCategory,
  1055→  1055→      retailVsService: [
  1056→  1056→        { type: 'Retail', revenue: retailRevenue },
  1057→  1057→        { type: 'Service', revenue: serviceRevenue },
  1058→  1058→      ],
  1059→  1059→      monthlyRetailTrend: Object.entries(monthlyRetail)
  1060→  1060→        .sort(([a], [b]) => a.localeCompare(b))
  1061→  1061→        .map(([month, data]) => ({ month, ...data })),

  1063→  1063→    details: {

  1065→  1065→      topPerformers,
  1066→  1066→      slowMoving,
  1067→  1067→    },
  1068→  1068→  }
  1069→  1069→}
