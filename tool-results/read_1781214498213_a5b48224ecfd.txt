     5→     5→function getDateRange(from?: string, to?: string) {
     6→     6→  const dateFrom = from ? new Date(from) : new Date(new Date().getFullYear(), new Date().getMonth(), 1)
     7→     7→  const dateTo = to ? new Date(to) : new Date()
     8→     8→  dateTo.setHours(23, 59, 59, 999)
     9→     9→  dateFrom.setHours(0, 0, 0, 0)
    10→    10→  return { dateFrom, dateTo }
    11→    11→}

    13→    13→// 1. Revenue Report
    14→    14→async function getRevenueReport(dateFrom: Date, dateTo: Date) {
    15→    15→  const invoices = await db.invoice.findMany({
    16→    16→    where: {
    17→    17→      createdAt: { gte: dateFrom, lte: dateTo },
    18→    18→      paymentStatus: { in: ['paid', 'partial'] },
    19→    19→    },
    20→    20→    include: { items: true },
    21→    21→  })
    22→    22→
    23→    23→  const totalRevenue = invoices.reduce((s, i) => s + i.totalAmount, 0)
    24→    24→  const totalTax = invoices.reduce((s, i) => s + i.tax, 0)
    25→    25→  const totalDiscount = invoices.reduce((s, i) => s + i.discount, 0)
    26→    26→  const avgInvoiceValue = invoices.length > 0 ? totalRevenue / invoices.length : 0
    27→    27→
    28→    28→  const paymentMethods: Record<string, number> = {}
    29→    29→  invoices.forEach((i) => {
    30→    30→    paymentMethods[i.paymentMethod] = (paymentMethods[i.paymentMethod] || 0) + i.totalAmount
    31→    31→  })
    32→    32→
    33→    33→  const dailyRevenue: Record<string, number> = {}
    34→    34→  invoices.forEach((i) => {
    35→    35→    const day = format(i.createdAt, 'yyyy-MM-dd')
    36→    36→    dailyRevenue[day] = (dailyRevenue[day] || 0) + i.totalAmount
    37→    37→  })
    38→    38→
    39→    39→  const serviceRevenue: Record<string, { name: string; revenue: number; count: number }> = {}
    40→    40→  invoices.forEach((inv) => {
    41→    41→    inv.items.forEach((item) => {
    42→    42→      if (item.itemType === 'service') {
    43→    43→        if (!serviceRevenue[item.name]) {
    44→    44→          serviceRevenue[item.name] = { name: item.name, revenue: 0, count: 0 }
    45→    45→        }
    46→    46→        serviceRevenue[item.name].revenue += item.total
    47→    47→        serviceRevenue[item.name].count += item.quantity
    48→    48→      }
    49→    49→    })
    50→    50→  })
    51→    51→
    52→    52→  const topServices = Object.values(serviceRevenue)
    53→    53→    .sort((a, b) => b.revenue - a.revenue)
    54→    54→    .slice(0, 10)
    55→    55→
    56→    56→  const revenueByType: Record<string, number> = {}
    57→    57→  invoices.forEach((inv) => {
    58→    58→    revenueByType[inv.type] = (revenueByType[inv.type] || 0) + inv.totalAmount
    59→    59→  })
    60→    60→
    61→    61→  return {
    62→    62→    summary: {
    63→    63→      totalRevenue,
    64→    64→      totalTax,
    65→    65→      totalDiscount,
    66→    66→      avgInvoiceValue,
    67→    67→      invoiceCount: invoices.length,
    68→    68→      periodStart: format(dateFrom, 'yyyy-MM-dd'),
    69→    69→      periodEnd: format(dateTo, 'yyyy-MM-dd'),
    70→    70→    },
    71→    71→    charts: {
    72→    72→      dailyRevenue: Object.entries(dailyRevenue)
    73→    73→        .sort(([a], [b]) => a.localeCompare(b))
    74→    74→        .map(([date, amount]) => ({ date, amount })),
    75→    75→      paymentMethods: Object.entries(paymentMethods).map(([method, amount]) => ({
    76→    76→        method,
    77→    77→        amount,
    78→    78→      })),
    79→    79→      revenueByType: Object.entries(revenueByType).map(([type, amount]) => ({ type, amount })),
    80→    80→    },
    81→    81→    details: {
    82→    82→      topServices,
    83→    83→    },
    84→    84→  }
    85→    85→}

    87→    87→// 2. Appointment Report
    88→    88→async function getAppointmentReport(dateFrom: Date, dateTo: Date) {
    89→    89→  const appointments = await db.appointment.findMany({
    90→    90→    where: { date: { gte: dateFrom, lte: dateTo } },
    91→    91→    include: {
    92→    92→      staff: { select: { firstName: true, lastName: true } },
    93→    93→      services: { include: { service: { select: { name: true } } } },
    94→    94→    },
    95→    95→  })
    96→    96→
    97→    97→  const total = appointments.length
    98→    98→  const statusBreakdown: Record<string, number> = {}
    99→    99→  appointments.forEach((a) => {
   100→   100→    statusBreakdown[a.status] = (statusBreakdown[a.status] || 0) + 1
   101→   101→  })
   102→   102→
   103→   103→  const hourCounts: Record<number, number> = {}
   104→   104→  appointments.forEach((a) => {
   105→   105→    const hour = parseInt(a.startTime.split(':')[0], 10)
   106→   106→    hourCounts[hour] = (hourCounts[hour] || 0) + 1
   107→   107→  })
   108→   108→
   109→   109→  const staffBookings: Record<string, { name: string; bookings: number; completed: number }> = {}
   110→   110→  appointments.forEach((a) => {
   111→   111→    const name = `${a.staff.firstName} ${a.staff.lastName}`
   112→   112→    if (!staffBookings[a.staffId]) {
   113→   113→      staffBookings[a.staffId] = { name, bookings: 0, completed: 0 }
   114→   114→    }
   115→   115→    staffBookings[a.staffId].bookings++
   116→   116→    if (a.status === 'completed') staffBookings[a.staffId].completed++
   117→   117→  })
   118→   118→
   119→   119→  const dailyTrend: Record<string, { total: number; completed: number; cancelled: number }> = {}
   120→   120→  appointments.forEach((a) => {
   121→   121→    const day = format(a.date, 'yyyy-MM-dd')
   122→   122→    if (!dailyTrend[day]) dailyTrend[day] = { total: 0, completed: 0, cancelled: 0 }
   123→   123→    dailyTrend[day].total++
   124→   124→    if (a.status === 'completed') dailyTrend[day].completed++
   125→   125→    if (a.status === 'cancelled') dailyTrend[day].cancelled++
   126→   126→  })
   127→   127→
   128→   128→  const typeBreakdown: Record<string, number> = {}
   129→   129→  appointments.forEach((a) => {
   130→   130→    typeBreakdown[a.type] = (typeBreakdown[a.type] || 0) + 1
   131→   131→  })
   132→   132→
   133→   133→  const completedRate = total > 0 ? ((statusBreakdown['completed'] || 0) / total) * 100 : 0
   134→   134→  const noShowRate = total > 0 ? ((statusBreakdown['no_show'] || 0) / total) * 100 : 0
   135→   135→  const cancellationRate = total > 0 ? ((statusBreakdown['cancelled'] || 0) / total) * 100 : 0
   136→   136→
   137→   137→  return {
   138→   138→    summary: {
   139→   139→      totalAppointments: total,
   140→   140→      completedRate: Math.round(completedRate * 10) / 10,
   141→   141→      noShowRate: Math.round(noShowRate * 10) / 10,
   142→   142→      cancellationRate: Math.round(cancellationRate * 10) / 10,
   143→   143→    },
   144→   144→    charts: {
   145→   145→      statusBreakdown: Object.entries(statusBreakdown).map(([status, count]) => ({
   146→   146→        status,
   147→   147→        count,
   148→   148→      })),
   149→   149→      peakHours: Object.entries(hourCounts)
   150→   150→        .sort(([a], [b]) => Number(a) - Number(b))
   151→   151→        .map(([hour, count]) => ({
   152→   152→          hour: `${hour}:00`,
   153→   153→          count,
   154→   154→        })),
   155→   155→      dailyTrend: Object.entries(dailyTrend)
   156→   156→        .sort(([a], [b]) => a.localeCompare(b))
   157→   157→        .map(([date, data]) => ({ date, ...data })),
   158→   158→      typeBreakdown: Object.entries(typeBreakdown).map(([type, count]) => ({ type, count })),
   159→   159→    },
   160→   160→    details: {
   161→   161→      staffBookings: Object.values(staffBookings).sort((a, b) => b.bookings - a.bookings),
   162→   162→    },
   163→   163→  }
   164→   164→}

   178→   178→  })
   179→   179→
   180→   180→  const feedback = await db.feedback.findMany({
   181→   181→    where: { createdAt: { gte: dateFrom, lte: dateTo }, staffId: { not: null } },
   182→   182→  })
   183→   183→
   184→   184→  const staffRatings: Record<string, { total: number; count: number }> = {}
   185→   185→  feedback.forEach((f) => {
   186→   186→    if (!f.staffId) return
   187→   187→    if (!staffRatings[f.staffId]) staffRatings[f.staffId] = { total: 0, count: 0 }
   188→   188→    staffRatings[f.staffId].total += f.rating
   189→   189→    staffRatings[f.staffId].count++
   190→   190→  })
   191→   191→
   192→   192→  const staffMetrics = staffMembers.map((s) => {
   193→   193→    const appointmentsCompleted = s.appointments.length

   195→   195→    const commissionsEarned = s.commissions.reduce((sum, c) => sum + c.amount, 0)
   196→   196→    const commissionsPaid = s.commissions
   197→   197→      .filter((c) => c.status === 'paid')
   198→   198→      .reduce((sum, c) => sum + c.amount, 0)

   200→   200→      appointmentsCompleted > 0 ? totalRevenue / appointmentsCompleted : 0
   201→   201→    const avgRating = staffRatings[s.id]

   203→   203→      : 0
   204→   204→
   205→   205→    return {
   206→   206→      name: `${s.firstName} ${s.lastName}`,
   207→   207→      role: s.role,
   208→   208→      specialization: s.specialization || '—',
   209→   209→      appointmentsCompleted,
   210→   210→      totalRevenue,
   211→   211→      commissionsEarned,
   212→   212→      commissionsPaid,
   213→   213→      avgRevenuePerAppointment: Math.round(avgRevenuePerAppointment),
   214→   214→      commissionRate: s.commissionRate,
   215→   215→      avgRating,
   216→   216→    }
   217→   217→  })
   218→   218→
   219→   219→  const totalRevenue = staffMetrics.reduce((s, m) => s + m.totalRevenue, 0)
   220→   220→  const totalCommissions = staffMetrics.reduce((s, m) => s + m.commissionsEarned, 0)
   221→   221→  const totalAppointments = staffMetrics.reduce((s, m) => s + m.appointmentsCompleted, 0)
   222→   222→
   223→   223→  return {
   224→   224→    summary: {
   225→   225→      totalStaff: staffMembers.length,
   226→   226→      totalAppointments,
   227→   227→      totalRevenue,
   228→   228→      totalCommissions,
   229→   229→      avgRevenuePerStaff: staffMembers.length > 0 ? Math.round(totalRevenue / staffMembers.length) : 0,
   230→   230→    },
   231→   231→    charts: {
   232→   232→      revenueByStaff: staffMetrics
   233→   233→        .sort((a, b) => b.totalRevenue - a.totalRevenue)
   234→   234→        .slice(0, 10)
   235→   235→        .map((m) => ({ name: m.name, revenue: m.totalRevenue })),
   236→   236→      appointmentsByStaff: staffMetrics
   237→   237→        .sort((a, b) => b.appointmentsCompleted - a.appointmentsCompleted)
   238→   238→        .slice(0, 10)
   239→   239→        .map((m) => ({ name: m.name, appointments: m.appointmentsCompleted })),
   240→   240→    },
   241→   241→    details: {
   242→   242→      staffMetrics,
   243→   243→    },
   244→   244→  }
   245→   245→}

   248→   248→async function getCustomerReport(dateFrom: Date, dateTo: Date) {

   250→   250→    where: { createdAt: { gte: dateFrom, lte: dateTo } },

   257→   257→  const returningCustomers = allCustomers.filter((c) => c.totalVisits > 1).length
   258→   258→  const retentionRate = totalCustomers > 0 ? (returningCustomers / totalCustomers) * 100 : 0
   259→   259→
   260→   260→  const genderDist: Record<string, number> = {}
   261→   261→  allCustomers.forEach((c) => {

   264→   264→  })
   265→   265→
   266→   266→  const tierDist: Record<string, number> = {}
   267→   267→  allCustomers.forEach((c) => {
   268→   268→    tierDist[c.loyaltyTier] = (tierDist[c.loyaltyTier] || 0) + 1
   269→   269→  })
   270→   270→
   271→   271→  const ltvBuckets: Record<string, number> = {
   272→   272→    'PKR 0-5K': 0,
   273→   273→    'PKR 5K-15K': 0,
   274→   274→    'PKR 15K-30K': 0,
   275→   275→    'PKR 30K-50K': 0,
   276→   276→    'PKR 50K+': 0,
   277→   277→  }
   278→   278→  allCustomers.forEach((c) => {
   279→   279→    if (c.lifetimeValue < 5000) ltvBuckets['PKR 0-5K']++
   280→   280→    else if (c.lifetimeValue < 15000) ltvBuckets['PKR 5K-15K']++
   281→   281→    else if (c.lifetimeValue < 30000) ltvBuckets['PKR 15K-30K']++
   282→   282→    else if (c.lifetimeValue < 50000) ltvBuckets['PKR 30K-50K']++
   283→   283→    else ltvBuckets['PKR 50K+']++
   284→   284→  })
   285→   285→
   286→   286→  const totalLifetimeValue = allCustomers.reduce((s, c) => s + c.lifetimeValue, 0)
   287→   287→  const avgLifetimeValue = totalCustomers > 0 ? totalLifetimeValue / totalCustomers : 0
   288→   288→
   289→   289→  const monthlyNew: Record<string, number> = {}
   290→   290→  customers.forEach((c) => {
   291→   291→    const month = format(c.createdAt, 'yyyy-MM')
   292→   292→    monthlyNew[month] = (monthlyNew[month] || 0) + 1
   293→   293→  })
   294→   294→
   295→   295→  return {
   296→   296→    summary: {
   297→   297→      totalCustomers,
   298→   298→      newCustomers,
   299→   299→      returningCustomers,
   300→   300→      retentionRate: Math.round(retentionRate * 10) / 10,
   301→   301→      totalLifetimeValue,
   302→   302→      avgLifetimeValue: Math.round(avgLifetimeValue),
   303→   303→    },
   304→   304→    charts: {
   305→   305→      genderDistribution: Object.entries(genderDist).map(([gender, count]) => ({ gender, count })),
   306→   306→      tierDistribution: Object.entries(tierDist).map(([tier, count]) => ({ tier, count })),
   307→   307→      ltvDistribution: Object.entries(ltvBuckets).map(([range, count]) => ({ range, count })),
   308→   308→      monthlyNewCustomers: Object.entries(monthlyNew)
   309→   309→        .sort(([a], [b]) => a.localeCompare(b))
   310→   310→        .map(([month, count]) => ({ month, count })),
   311→   311→    },
   312→   312→    details: {
   313→   313→      topCustomers: allCustomers
   314→   314→        .sort((a, b) => b.lifetimeValue - a.lifetimeValue)
   315→   315→        .slice(0, 15)

   322→   322→          points: c.loyaltyPoints,
   323→   323→        })),
   324→   324→    },
   325→   325→  }

   329→   329→async function getInventoryReport(dateFrom: Date, dateTo: Date) {
   330→   330→  const inventoryItems = await db.inventoryItem.findMany({
   331→   331→    include: {
   332→   332→      product: {
   333→   333→        include: { category: true },
   334→   334→      },
   335→   335→      branch: { select: { name: true } },
   336→   336→    },
   337→   337→  })
   338→   338→
   339→   339→  void dateFrom
   340→   340→  void dateTo
   341→   341→
   342→   342→  const lowStockItems = inventoryItems.filter(
   343→   343→    (i) => i.quantity <= i.product.reorderLevel
   344→   344→  )
   345→   345→
   346→   346→  const totalItems = inventoryItems.length
   347→   347→  const totalQuantity = inventoryItems.reduce((s, i) => s + i.quantity, 0)
   348→   348→  const totalValue = inventoryItems.reduce(
   349→   349→    (s, i) => s + i.quantity * i.product.costPrice,
   350→   350→    0
   351→   351→  )
   352→   352→  const totalSellingValue = inventoryItems.reduce(
   353→   353→    (s, i) => s + i.quantity * i.product.sellingPrice,
   354→   354→    0
   355→   355→  )
   356→   356→
   357→   357→  const categoryBreakdown: Record<string, { quantity: number; value: number }> = {}
   358→   358→  inventoryItems.forEach((i) => {
   359→   359→    const cat = i.product.category?.name || 'Uncategorized'
   360→   360→    if (!categoryBreakdown[cat]) categoryBreakdown[cat] = { quantity: 0, value: 0 }
   361→   361→    categoryBreakdown[cat].quantity += i.quantity
   362→   362→    categoryBreakdown[cat].value += i.quantity * i.product.costPrice
   363→   363→  })
   364→   364→
   365→   365→  const branchBreakdown: Record<string, { items: number; quantity: number; value: number }> = {}
   366→   366→  inventoryItems.forEach((i) => {
   367→   367→    const branch = i.branch.name
   368→   368→    if (!branchBreakdown[branch]) branchBreakdown[branch] = { items: 0, quantity: 0, value: 0 }
   369→   369→    branchBreakdown[branch].items++
   370→   370→    branchBreakdown[branch].quantity += i.quantity
   371→   371→    branchBreakdown[branch].value += i.quantity * i.product.costPrice
   372→   372→  })
   373→   373→
   374→   374→  const thirtyDaysFromNow = new Date()
   375→   375→  thirtyDaysFromNow.setDate(thirtyDaysFromNow.getDate() + 30)
   376→   376→  const expiringSoon = inventoryItems.filter(
   377→   377→    (i) => i.expiryDate && i.expiryDate <= thirtyDaysFromNow
   378→   378→  )
   379→   379→
   380→   380→  return {
   381→   381→    summary: {
   382→   382→      totalProducts: totalItems,
   383→   383→      totalQuantity,
   384→   384→      totalValue,
   385→   385→      totalSellingValue,
   386→   386→      lowStockCount: lowStockItems.length,
   387→   387→      expiringSoonCount: expiringSoon.length,
   388→   388→      potentialProfit: totalSellingValue - totalValue,
   389→   389→    },
   390→   390→    charts: {
   391→   391→      categoryBreakdown: Object.entries(categoryBreakdown).map(([category, data]) => ({
   392→   392→        category,
   393→   393→        quantity: data.quantity,
   394→   394→        value: data.value,
   395→   395→      })),
   396→   396→      branchBreakdown: Object.entries(branchBreakdown).map(([branch, data]) => ({
   397→   397→        branch,
   398→   398→        items: data.items,
   399→   399→        quantity: data.quantity,
   400→   400→        value: data.value,
   401→   401→      })),
   402→   402→      stockLevels: inventoryItems
   403→   403→        .sort((a, b) => a.quantity - b.quantity)
   404→   404→        .slice(0, 15)
   405→   405→        .map((i) => ({
   406→   406→          name: i.product.name,
   407→   407→          quantity: i.quantity,
   408→   408→          reorderLevel: i.product.reorderLevel,
   409→   409→        })),
   410→   410→    },
   411→   411→    details: {
   412→   412→      lowStockItems: lowStockItems.map((i) => ({
   413→   413→        name: i.product.name,
   414→   414→        sku: i.product.sku,
   415→   415→        branch: i.branch.name,
   416→   416→        quantity: i.quantity,
   417→   417→        reorderLevel: i.product.reorderLevel,
   418→   418→        category: i.product.category?.name || '—',
   419→   419→      })),
   420→   420→      expiringSoon: expiringSoon.map((i) => ({
   421→   421→        name: i.product.name,
   422→   422→        branch: i.branch.name,
   423→   423→        quantity: i.quantity,
   424→   424→        expiryDate: i.expiryDate ? format(i.expiryDate, 'yyyy-MM-dd') : '—',

   436→   436→
   437→   437→  const expenses = await db.expense.findMany({
   438→   438→    where: { date: { gte: dateFrom, lte: dateTo } },

   445→   445→  const totalExpenses = expenses.reduce((s, e) => s + e.amount, 0)
   446→   446→  const grossProfit = totalRevenue - totalExpenses

   448→   448→

   454→   454→  const revenueByType: Record<string, number> = {}
   455→   455→  invoices.forEach((inv) => {

   457→   457→  })
   458→   458→
   459→   459→  const monthlyPL: Record<string, { revenue: number; expenses: number; profit: number }> = {}

   469→   469→  })
   470→   470→  Object.values(monthlyPL).forEach((m) => {
   471→   471→    m.profit = m.revenue - m.expenses
   472→   472→  })
   473→   473→
   474→   474→  const itemRevenue: Record<string, { type: string; revenue: number; count: number }> = {}
   475→   475→  invoices.forEach((inv) => {

   477→   477→      if (!itemRevenue[item.name]) {
   478→   478→        itemRevenue[item.name] = { type: item.itemType, revenue: 0, count: 0 }
   479→   479→      }
   480→   480→      itemRevenue[item.name].revenue += item.total
   481→   481→      itemRevenue[item.name].count += item.quantity
   482→   482→    })
   483→   483→  })
   484→   484→
   485→   485→  return {
   486→   486→    summary: {

   498→   498→        .sort(([a], [b]) => a.localeCompare(b))
   499→   499→        .map(([month, data]) => ({ month, ...data })),
   500→   500→      expenseCategories: Object.entries(expenseCategories).map(([category, amount]) => ({
   501→   501→        category,

   503→   503→      })),
   504→   504→      revenueByType: Object.entries(revenueByType).map(([type, amount]) => ({ type, amount })),
