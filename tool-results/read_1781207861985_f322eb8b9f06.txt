   160→function formatCurrency(amount: number): string {
   161→  return `PKR ${amount.toLocaleString()}`
   162→}
   163→
   164→function formatNumber(num: number): string {
   165→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`
   166→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`
   167→  return num.toLocaleString()
   168→}
   169→
   170→function getStatusBadgeStyle(status: string): string {
   171→  switch (status) {
   172→    case 'completed':
   173→      return 'bg-emerald-100 text-emerald-700 border-emerald-200 dark:bg-emerald-900/30 dark:text-emerald-300 dark:border-emerald-800'
   174→    case 'confirmed':
   175→      return 'bg-blue-100 text-blue-700 border-blue-200 dark:bg-blue-900/30 dark:text-blue-300 dark:border-blue-800'
   176→    case 'in_progress':
   177→      return 'bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/30 dark:text-amber-300 dark:border-amber-800'
   178→    case 'cancelled':
   179→      return 'bg-red-100 text-red-700 border-red-200 dark:bg-red-900/30 dark:text-red-300 dark:border-red-800'
   180→    case 'no_show':
   181→      return 'bg-orange-100 text-orange-700 border-orange-200 dark:bg-orange-900/30 dark:text-orange-300 dark:border-orange-800'
   182→    default: // scheduled
   183→      return 'bg-violet-100 text-violet-700 border-violet-200 dark:bg-violet-900/30 dark:text-violet-300 dark:border-violet-800'
   184→  }
   185→}

   187→function getStatusLabel(status: string): string {
   188→  const map: Record<string, string> = {
   189→    scheduled: 'Scheduled',
   190→    confirmed: 'Confirmed',
   191→    in_progress: 'In Progress',
   192→    completed: 'Completed',
   193→    cancelled: 'Cancelled',
   194→    no_show: 'No Show',
   195→  }
   196→  return map[status] || status
   197→}

   286→
   287→  useEffect(() => {
   288→    let cancelled = false
   289→
   290→    async function fetchData() {
   291→      try {
   292→        setLoading(true)
   293→        const res = await fetch('/api/dashboard')
   294→        if (!res.ok) throw new Error('Failed to fetch dashboard data')
   295→        const json = await res.json()
   296→        if (!cancelled && json.success) {
   297→          setData(json.data)
   298→        } else if (!cancelled && !json.success) {
   299→          setError(json.error || 'Unknown error')
   300→        }
   301→      } catch (err) {
   302→        if (!cancelled) {
   303→          setError(err instanceof Error ? err.message : 'Failed to fetch')
   304→        }
   305→      } finally {
   306→        if (!cancelled) {
   307→          setLoading(false)
   308→        }
   309→      }
   310→    }
   311→
   312→    fetchData()

   314→  }, [])

   316→  if (loading) return <DashboardSkeleton />
