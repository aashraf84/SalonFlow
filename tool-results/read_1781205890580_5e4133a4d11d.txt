    16→    16→// ─── Device Fingerprint Generator ────────────────────────────────────────────
    17→    17→function generateDeviceId(): string {
    18→    18→  if (typeof window === 'undefined') return ''
    19→    19→  const canvas = document.createElement('canvas')
    20→    20→  const ctx = canvas.getContext('2d')
    21→    21→  if (ctx) {
    22→    22→    ctx.textBaseline = 'top'
    23→    23→    ctx.font = '14px Arial'
    24→    24→    ctx.fillText('SalonFlow-Device-Fingerprint', 2, 2)
    25→    25→  }
    26→    26→  const canvasData = canvas.toDataURL()
    27→    27→  const nav = navigator
    28→    28→  const screen = window.screen
    29→    29→  const raw = [
    30→    30→    nav.userAgent,
    31→    31→    nav.language,
    32→    32→    screen.width + 'x' + screen.height,
    33→    33→    screen.colorDepth,
    34→    34→    new Date().getTimezoneOffset(),
    35→    35→    nav.hardwareConcurrency || 'n/a',
    36→    36→    canvasData.slice(-50),
    37→    37→  ].join('||')
    38→    38→  // Simple hash
    39→    39→  let hash = 0
    40→    40→  for (let i = 0; i < raw.length; i++) {
    41→    41→    const char = raw.charCodeAt(i)
    42→    42→    hash = ((hash << 5) - hash) + char
    43→    43→    hash = hash & hash
    44→    44→  }
    45→    45→  return 'SF-DEV-' + Math.abs(hash).toString(36).toUpperCase() + '-' + Date.now().toString(36).toUpperCase()
    46→    46→}

    48→    48→// ─── Live Clock Component ────────────────────────────────────────────────────
    49→    49→function LiveClock() {
    50→    50→  const [time, setTime] = useState(new Date())
    51→    51→
    52→    52→  useEffect(() => {
    53→    53→    const interval = setInterval(() => setTime(new Date()), 1000)
    54→    54→    return () => clearInterval(interval)
    55→    55→  }, [])
    56→    56→
    57→    57→  const timeStr = time.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: true })
    58→    58→  const dateStr = time.toLocaleDateString('en-US', { weekday: 'long', month: 'long', day: 'numeric' })
    59→    59→
    60→    60→  return (
    61→    61→    <motion.div
    62→    62→      initial={{ opacity: 0, y: -20 }}
    63→    63→      animate={{ opacity: 1, y: 0 }}
    64→    64→      transition={{ delay: 0.2, duration: 0.8 }}
    65→    65→      className="text-center mb-8 select-none"
    66→    66→    >
    67→    67→      <p className="text-7xl sm:text-8xl font-extralight text-white tracking-tight drop-shadow-2xl" style={{ fontFeatureSettings: '"tnum"' }}>
    68→    68→        {timeStr}
    69→    69→      </p>
    70→    70→      <p className="text-lg sm:text-xl font-light text-white/70 mt-1 tracking-wide">
    71→    71→        {dateStr}
    72→    72→      </p>
    73→    73→    </motion.div>
    74→    74→  )
    75→    75→}
