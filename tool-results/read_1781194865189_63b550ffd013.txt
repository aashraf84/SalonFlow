   223→   223→            ? { ...item, quantity: item.quantity + 1 }

   226→   226→      }

   228→   228→        id: service.id,
   229→   229→        itemType: 'service',
   230→   230→        name: service.name,
   231→   231→        unitPrice: service.price,
   232→   232→        quantity: 1,

   247→   247→      prev.map((item) => (item.id === id ? { ...item, quantity } : item))

   295→   295→          customerId: selectedCustomerId || null,
   296→   296→          type: 'service',
   297→   297→          subtotal,
   298→   298→          tax,
   299→   299→          discount: discountValue,
   300→   300→          totalAmount: grandTotal,
   301→   301→          paymentMethod,
   302→   302→          paymentStatus: 'paid',
   303→   303→          items: cart.map((item) => ({
   304→   304→            itemType: item.itemType,
   305→   305→            itemId: item.id,
   306→   306→            name: item.name,
   307→   307→            quantity: item.quantity,
   308→   308→            unitPrice: item.unitPrice,
   309→   309→            discount: 0,
   310→   310→            total: item.unitPrice * item.quantity,
   311→   311→          })),
   312→   312→        }),

   314→   314→      const json = await res.json()

   316→   316→        setCart([])
   317→   317→        setDiscountPercent('0')
   318→   318→        setSelectedCustomerId('')
   319→   319→        setPaymentMethod('cash')
   320→   320→        fetchInvoices()
   321→   321→        setActiveTab('invoices')
   322→   322→      }
   323→   323→    } catch (err) {
   324→   324→      console.error('Failed to create invoice:', err)
   325→   325→    } finally {
   326→   326→      setSubmitting(false)
   327→   327→    }
   328→   328→  }
   329→   329→
   330→   330→  // Filter services for POS
   331→   331→  const allServices = categories.flatMap((cat) =>
   332→   332→    cat.services.map((svc) => ({ ...svc, categoryName: cat.name, categoryId: cat.id }))

   337→   337→      svc.categoryName.toLowerCase().includes(searchQuery.toLowerCase())

   339→   339→    return matchesSearch && matchesCategory
   340→   340→  })
   341→   341→
   342→   342→  if (loading) {

   347→   347→          <div className="space-y-2">
   348→   348→            <Skeleton className="h-5 w-48" />

   350→   350→          </div>
