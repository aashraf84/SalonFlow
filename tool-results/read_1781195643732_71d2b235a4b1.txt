    76→// ─── Main POS Component ─────────────────────────────────────────────────────
    77→export function PosModule() {
    78→  // Data
    79→  const [invoices, setInvoices] = useState<InvoiceType[]>([])
    80→  const [categories, setCategories] = useState<CategoryType[]>([])
    81→  const [customers, setCustomers] = useState<CustomerType[]>([])
    82→  const [loading, setLoading] = useState(true)
    83→
    84→  // POS State
    85→  const [cart, setCart] = useState<CartItem[]>([])
    86→  const [searchQuery, setSearchQuery] = useState('')
    87→  const [selectedCategory, setSelectedCategory] = useState<string>('all')
    88→  const [selectedCustomer, setSelectedCustomer] = useState<CustomerType | null>(null)
    89→  const [showCustomerPicker, setShowCustomerPicker] = useState(false)
    90→  const [customerSearch, setCustomerSearch] = useState('')
    91→  const [paymentMethod, setPaymentMethod] = useState('cash')
    92→  const [discountPercent, setDiscountPercent] = useState(0)
    93→  const [customDiscount, setCustomDiscount] = useState('')
    94→  const [submitting, setSubmitting] = useState(false)
    95→  const [showCheckout, setShowCheckout] = useState(false)
    96→  const [showReceipt, setShowReceipt] = useState(false)
    97→  const [completedInvoice, setCompletedInvoice] = useState<InvoiceType | null>(null)
    98→  const [viewMode, setViewMode] = useState<'pos' | 'history'>('pos')
    99→
   100→  // ─── Data Fetching ──────────────────────────────────────────────────────
   101→  const fetchInvoices = useCallback(async () => {
   102→    try {
   103→      const res = await fetch('/api/invoices?take=100')
   104→      const json = await res.json()
   105→      if (json.success) setInvoices(json.data)
   106→    } catch { toast.error('Failed to load invoices') }
   107→  }, [])
   108→
   109→  const fetchServices = useCallback(async () => {
   110→    try {
   111→      const res = await fetch('/api/services')
   112→      const json = await res.json()
   113→      if (json.success) setCategories(json.data)
   114→    } catch { toast.error('Failed to load services') }
   115→  }, [])
   116→
   117→  const fetchCustomers = useCallback(async () => {
   118→    try {
   119→      const res = await fetch('/api/customers?take=200')
   120→      const json = await res.json()
   121→      if (json.success) setCustomers(json.data)
   122→    } catch { toast.error('Failed to load customers') }
   123→  }, [])
   124→
   125→  useEffect(() => {
   126→    const load = async () => {
   127→      setLoading(true)
   128→      await Promise.all([fetchInvoices(), fetchServices(), fetchCustomers()])
   129→      setLoading(false)
   130→    }
   131→    load()
   132→  }, [fetchInvoices, fetchServices, fetchCustomers])
   133→
   134→  // ─── Cart Logic ─────────────────────────────────────────────────────────
   135→  const allServices = useMemo(() =>
   136→    categories.flatMap(cat => cat.services.map(svc => ({ ...svc, categoryName: cat.name, categoryId: cat.id }))),
   137→    [categories]
   138→  )
   139→
   140→  const filteredServices = useMemo(() =>
   141→    allServices.filter(svc => {
   142→      const q = searchQuery.toLowerCase()
   143→      const matchesSearch = !q || svc.name.toLowerCase().includes(q) || svc.categoryName.toLowerCase().includes(q)
   144→      const matchesCat = selectedCategory === 'all' || svc.categoryId === selectedCategory
   145→      return matchesSearch && matchesCat && svc.isActive
   146→    }),
   147→    [allServices, searchQuery, selectedCategory]
   148→  )
   149→
   150→  const addToCart = useCallback((service: ServiceType & { categoryName?: string; categoryId?: string }) => {
   151→    setCart(prev => {
   152→      const existing = prev.find(item => item.id === service.id)
   153→      if (existing) {
   154→        return prev.map(item =>
   155→          item.id === service.id ? { ...item, quantity: item.quantity + 1 } : item
   156→        )
   157→      }
   158→      return [...prev, {
   159→        id: service.id,
   160→        itemType: 'service',
   161→        name: service.name,
   162→        unitPrice: service.price,
   163→        quantity: 1,
   164→        duration: service.duration,
   165→        discount: 0,
   166→        category: service.categoryName
   167→      }]
   168→    })
   169→    toast.success(`${service.name} added`, { duration: 1200 })
   170→  }, [])
   171→
   172→  const removeFromCart = useCallback((id: string) => {
   173→    setCart(prev => prev.filter(item => item.id !== id))
   174→  }, [])
   175→
   176→  const updateQuantity = useCallback((id: string, delta: number) => {
   177→    setCart(prev => {
   178→      const item = prev.find(i => i.id === id)
   179→      if (!item) return prev
   180→      const newQty = item.quantity + delta
   181→      if (newQty < 1) return prev.filter(i => i.id !== id)
   182→      return prev.map(i => i.id === id ? { ...i, quantity: newQty } : i)
   183→    })
   184→  }, [])
   185→
   186→  const clearCart = useCallback(() => {
   187→    setCart([])
   188→    setDiscountPercent(0)
   189→    setCustomDiscount('')
   190→    setSelectedCustomer(null)
   191→  }, [])
   192→
   193→  // ─── Calculations ───────────────────────────────────────────────────────
   194→  const subtotal = useMemo(() => cart.reduce((sum, item) => sum + (item.unitPrice * item.quantity) - item.discount, 0), [cart])
   195→  const taxRate = 0.05
   196→  const tax = subtotal * taxRate
   197→  const discountValue = subtotal * discountPercent / 100
   198→  const grandTotal = subtotal + tax - discountValue
   199→  const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0)
   200→  const totalDuration = cart.reduce((sum, item) => sum + (item.duration || 0) * item.quantity, 0)
   201→
   202→  // Quick stats
   203→  const todayStr = new Date().toISOString().split('T')[0]
   204→  const todaySales = invoices.filter(inv => inv.createdAt.split('T')[0] === todayStr && inv.paymentStatus === 'paid').reduce((s, i) => s + i.totalAmount, 0)
   205→  const todayTransactions = invoices.filter(inv => inv.createdAt.split('T')[0] === todayStr && inv.paymentStatus === 'paid').length
   206→
   207→  // ─── Complete Sale ──────────────────────────────────────────────────────
   208→  const handleCompleteSale = async () => {
   209→    if (cart.length === 0) { toast.error('Cart is empty'); return }
   210→    const firstBranch = invoices.find(inv => inv.branch)?.branchId
   211→    if (!firstBranch) { toast.error('No branch found. Create a branch first.'); return }
   212→
   213→    setSubmitting(true)
   214→    try {
   215→      const res = await fetch('/api/invoices', {
   216→        method: 'POST',
   217→        headers: { 'Content-Type': 'application/json' },
   218→        body: JSON.stringify({
   219→          branchId: firstBranch,
   220→          customerId: selectedCustomer?.id || null,
   221→          type: 'service',
   222→          subtotal,
   223→          tax,
   224→          discount: discountValue,
   225→          totalAmount: grandTotal,
   226→          paymentMethod,
   227→          paymentStatus: 'paid',
   228→          items: cart.map(item => ({
   229→            itemType: item.itemType,
   230→            itemId: item.id,
   231→            name: item.name,
   232→            quantity: item.quantity,
   233→            unitPrice: item.unitPrice,
   234→            discount: item.discount,
   235→            total: (item.unitPrice * item.quantity) - item.discount,
   236→          })),
   237→        }),
   238→      })
   239→      const json = await res.json()
   240→      if (json.success) {
   241→        toast.success('Sale completed successfully!')
   242→        setCompletedInvoice(json.data)
   243→        setShowCheckout(false)
   244→        setShowReceipt(true)
   245→        fetchInvoices()
   246→      } else {
   247→        toast.error(json.error || 'Failed to complete sale')
   248→      }
   249→    } catch { toast.error('Failed to complete sale') }
   250→    finally { setSubmitting(false) }
   251→  }
   252→
   253→  // ─── Print Receipt ──────────────────────────────────────────────────────
   254→  const printReceipt = () => {
   255→    if (!completedInvoice) return
   256→    const c = completedInvoice.customer
   257→    const w = window.open('', '_blank')
   258→    if (!w) return
   259→    w.document.write(`<!DOCTYPE html><html><head><title>Receipt ${completedInvoice.invoiceNumber}</title>
   260→    <style>
   261→      @page { size: 80mm auto; margin: 0; }
   262→      * { margin: 0; padding: 0; box-sizing: border-box; }
   263→      body { font-family: 'Courier New', monospace; font-size: 11px; color: #000; width: 80mm; padding: 5mm; }
   264→      .center { text-align: center; }
   265→      .bold { font-weight: bold; }
   266→      .line { border-top: 1px dashed #000; margin: 5px 0; }
   267→      .double-line { border-top: 2px solid #000; margin: 5px 0; }
   268→      table { width: 100%; }
   269→      td { padding: 1px 0; }
   270→      td:last-child { text-align: right; }
   271→      .logo { font-size: 16px; font-weight: bold; letter-spacing: 1px; }
   272→      h2 { font-size: 13px; letter-spacing: 2px; }
   273→    </style></head><body>
   274→    <div class="center">
   275→      <p class="logo">✂ SALONFLOW OS</p>
   276→      <p>Beauty & Spa Management</p>
   277→      <p>DHA Phase 6, Karachi</p>
   278→      <p>Tel: +92-300-1234567</p>
   279→    </div>
   280→    <div class="double-line"></div>
   281→    <p class="center bold"><h2>RECEIPT</h2></p>
   282→    <p>Invoice: <span class="bold">${completedInvoice.invoiceNumber}</span></p>
   283→    <p>Date: ${new Date(completedInvoice.createdAt).toLocaleString()}</p>
   284→    <p>Cashier: Admin</p>
   285→    ${c ? `<p>Customer: ${c.firstName} ${c.lastName}</p><p>Phone: ${c.phone}</p>` : ''}
   286→    <div class="line"></div>
   287→    <table>
   288→      <tr><td class="bold">Item</td><td class="bold">Amt</td></tr>
   289→      ${(completedInvoice.items || []).map(item => `
   290→        <tr><td>${item.name} x${item.quantity}</td><td>${formatPKR(item.total)}</td></tr>
   291→      `).join('')}
   292→    </table>
   293→    <div class="line"></div>
   294→    <table>
   295→      <tr><td>Subtotal</td><td>${formatPKR(completedInvoice.subtotal)}</td></tr>
   296→      ${completedInvoice.tax > 0 ? `<tr><td>Tax (5%)</td><td>${formatPKR(completedInvoice.tax)}</td></tr>` : ''}
   297→      ${completedInvoice.discount > 0 ? `<tr><td>Discount</td><td>-${formatPKR(completedInvoice.discount)}</td></tr>` : ''}
   298→    </table>
   299→    <div class="double-line"></div>
   300→    <table>
   301→      <tr class="bold" style="font-size:14px"><td>TOTAL</td><td>${formatPKR(completedInvoice.totalAmount)}</td></tr>
   302→    </table>
   303→    <div class="double-line"></div>
   304→    <p>Payment: ${completedInvoice.paymentMethod.toUpperCase()}</p>
   305→    <div class="line"></div>
   306→    <div class="center" style="margin-top:8px;">
   307→      <p>Thank you for your visit!</p>
   308→      <p style="font-size:9px;margin-top:3px;">SalonFlow OS v1.0 | Powered by ApexFlow</p>
   309→    </div>
   310→    </body></html>`)
   311→    w.document.close()
   312→    setTimeout(() => w.print(), 300)
   313→  }
   314→
   315→  // ─── Loading State ──────────────────────────────────────────────────────
   316→  if (loading) {
   317→    return (
   318→      <div className="space-y-4">
   319→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   320→          <div className="lg:col-span-2 space-y-4">
   321→            <Skeleton className="h-16 rounded-2xl" />
   322→            <Skeleton className="h-96 rounded-2xl" />
   323→          </div>
   324→          <Skeleton className="h-[500px] rounded-2xl" />
   325→        </div>
   326→      </div>
   327→    )
   328→  }
   329→
   330→  return (
   331→    <div className="space-y-4 h-full">
   332→      {/* ─── Top Stats Bar ──────────────────────────────────────────────────── */}
   333→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   334→        <AnimatedMetricCard title="Today's Sales" value={formatPKR(todaySales)} subtitle={`${todayTransactions} transactions`} icon={DollarSign} accentGradient="from-emerald-500/20 to-teal-500/20" iconBg="bg-emerald-500/10 text-emerald-600 dark:text-emerald-400" delay={0} />
   335→        <AnimatedMetricCard title="Cart Total" value={formatPKR(grandTotal)} subtitle={`${totalItems} items`} icon={ShoppingCart} accentGradient="from-rose-500/20 to-pink-500/20" iconBg="bg-rose-500/10 text-rose-600 dark:text-rose-400" delay={0.05} />
   336→        <AnimatedMetricCard title="Est. Duration" value={`${Math.floor(totalDuration / 60)}h ${totalDuration % 60}m`} subtitle="Total service time" icon={Clock} accentGradient="from-blue-500/20 to-indigo-500/20" iconBg="bg-blue-500/10 text-blue-600 dark:text-blue-400" delay={0.1} />
   337→        <AnimatedMetricCard title="Avg. Transaction" value={formatPKR(invoices.length > 0 ? invoices.reduce((s, i) => s + i.totalAmount, 0) / invoices.length : 0)} subtitle="Per invoice" icon={TrendingUp} accentGradient="from-amber-500/20 to-orange-500/20" iconBg="bg-amber-500/10 text-amber-600 dark:text-amber-400" delay={0.15} />
   338→      </div>
   339→
   340→      {/* ─── Mode Toggle ────────────────────────────────────────────────────── */}
   341→      <div className="flex items-center gap-2">
   342→        <Button size="sm" variant={viewMode === 'pos' ? 'default' : 'outline'} className={cn('text-xs rounded-xl gap-1.5', viewMode === 'pos' && 'bg-tahoe-accent text-white')} onClick={() => setViewMode('pos')}>
   343→          <ShoppingCart className="size-3.5" /> Point of Sale
   344→        </Button>
   345→        <Button size="sm" variant={viewMode === 'history' ? 'default' : 'outline'} className={cn('text-xs rounded-xl gap-1.5', viewMode === 'history' && 'bg-tahoe-accent text-white')} onClick={() => setViewMode('history')}>
   346→          <Receipt className="size-3.5" /> Transaction History
   347→        </Button>
   348→      </div>
   349→
   350→      {viewMode === 'pos' ? (
   351→        /* ═══ POS LAYOUT ═══════════════════════════════════════════════════════ */
   352→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4" style={{ minHeight: 'calc(100vh - 320px)' }}>
   353→          {/* ─── LEFT: Product Grid ──────────────────────────────────────────── */}
   354→          <div className="lg:col-span-2 space-y-3">
   355→            {/* Search & Category Filter */}
   356→            <GlassCard className="p-3" hover3d={false} animateOnMount={false}>
   357→              <div className="flex flex-col sm:flex-row gap-2">
   358→                <div className="relative flex-1">
   359→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   360→                  <Input
   361→                    placeholder="Search services... (try 'haircut' or 'facial')"
   362→                    className="pl-10 h-10 text-sm rounded-xl bg-foreground/[0.02] border-border/50"
   363→                    value={searchQuery}
   364→                    onChange={e => setSearchQuery(e.target.value)}
   365→                  />
   366→                  {searchQuery && (
   367→                    <Button variant="ghost" size="sm" className="absolute right-1 top-1/2 -translate-y-1/2 h-7 w-7 p-0" onClick={() => setSearchQuery('')}>
   368→                      <X className="size-3" />
   369→                    </Button>
   370→                  )}
   371→                </div>
   372→                <div className="flex gap-1.5 overflow-x-auto pb-1 sm:pb-0">
   373→                  <Button size="sm" variant={selectedCategory === 'all' ? 'default' : 'outline'} className={cn('text-[10px] rounded-lg shrink-0 px-3', selectedCategory === 'all' && 'bg-tahoe-accent text-white')} onClick={() => setSelectedCategory('all')}>
   374→                    All
   375→                  </Button>
   376→                  {categories.map(cat => (
   377→                    <Button key={cat.id} size="sm" variant={selectedCategory === cat.id ? 'default' : 'outline'} className={cn('text-[10px] rounded-lg shrink-0 px-3 gap-1', selectedCategory === cat.id && 'bg-tahoe-accent text-white')} onClick={() => setSelectedCategory(cat.id)}>
   378→                      <span>{categoryIcons[cat.name] || '✨'}</span>
   379→                      {cat.name}
   380→                    </Button>
   381→                  ))}
   382→                </div>
   383→              </div>
   384→            </GlassCard>
   385→
   386→            {/* Service Grid */}
   387→            <div className="grid grid-cols-2 sm:grid-cols-3 xl:grid-cols-4 gap-2">
   388→              <AnimatePresence>
   389→                {filteredServices.map(service => {
   390→                  const inCart = cart.find(i => i.id === service.id)
   391→                  return (
   392→                    <motion.button
   393→                      key={service.id}
   394→                      layout
   395→                      initial={{ opacity: 0, scale: 0.9 }}
   396→                      animate={{ opacity: 1, scale: 1 }}
   397→                      exit={{ opacity: 0, scale: 0.9 }}
   398→                      whileHover={{ scale: 1.03, y: -2 }}
   399→                      whileTap={{ scale: 0.97 }}
   400→                      onClick={() => addToCart(service)}
   401→                      className={cn(
   402→                        'relative text-left p-3 rounded-xl border transition-all duration-200 group',
   403→                        inCart
   404→                          ? 'border-rose-500/50 bg-rose-500/5 shadow-md shadow-rose-500/10'
   405→                          : 'border-border/50 bg-card/60 hover:border-rose-500/30 hover:bg-rose-500/[0.03]'
   406→                      )}
   407→                    >
   408→                      {/* Category Badge */}
   409→                      <div className="flex items-center justify-between mb-2">
   410→                        <span className="text-[9px] font-semibold uppercase tracking-wider text-muted-foreground/60 truncate">{service.categoryName}</span>
   411→                        {service.isPackage && (
   412→                          <Badge className="h-4 text-[8px] bg-amber-500/10 text-amber-600 border-amber-500/20 border px-1.5">PKG</Badge>
   413→                        )}
   414→                      </div>
   415→
   416→                      {/* Service Name */}
   417→                      <p className="text-sm font-semibold leading-tight mb-1 line-clamp-2">{service.name}</p>
   418→
   419→                      {/* Price & Duration */}
   420→                      <div className="flex items-center justify-between mt-2">
   421→                        <span className="text-sm font-bold text-rose-600 dark:text-rose-400">{formatPKR(service.price)}</span>
   422→                        <span className="text-[10px] text-muted-foreground">{service.duration}m</span>
   423→                      </div>
   424→
   425→                      {/* Cart Indicator */}
   426→                      {inCart && (
   427→                        <motion.div
   428→                          initial={{ scale: 0 }}
   429→                          animate={{ scale: 1 }}
   430→                          className="absolute -top-1.5 -right-1.5 flex size-5 items-center justify-center rounded-full bg-rose-500 text-white text-[10px] font-bold shadow-lg"
   431→                        >
   432→                          {inCart.quantity}
   433→                        </motion.div>
   434→                      )}
   435→
